//private acces modifier
class Person{
    private String name;
    private int age;

    public Person(String name, int age) {
        this.name = name;
        this.age = age;
    }
    private void seceretMethod() {
        System.out.println("This is a private seceret method");
    }
    public void callSeceret() {
        seceretMethod();
    }
    public void display() {
        System.out.println("MY name is: " + name);
        System.out.println("My age is: " + age);
    }
    public static void main(String[] args) {
        Person p = new Person("Singh", 21);
        p.name = "Atul";
        p.age = 21;
        p.display();
        p.seceretMethod();
        p.callSeceret();
    }
}

package Protected;

// Protected access modifier

class Animal {
    protected String name = "Tiger";
    protected void sound(){
        System.out.println("Hello");
    }
}

class Bird extends Animal{
    void makeSound(){
        sound();
        System.out.println("Bird class access");
    }
}

public class ProtectedModifier{
    public static void main(String[] args) {
        Animal a = new Animal();
        System.out.println("the name is "+a.name);
        a.sound();

        Bird b = new Bird();
        System.out.println("the tiger access through bird class  "+b.name);
        b.sound();
        b.makeSound();
    }
}

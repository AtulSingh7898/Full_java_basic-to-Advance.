    public static void main(String[] args) {
        int[] arr = { 10, 20, 30, 40, 50 };
        int insertIndex = 2;
        int insertValue = 100;
        for (int i = arr.length - 1; i > insertIndex; i--) {
            arr[i] = arr[i - 1];

        }
        arr[insertIndex]=insertValue;

        System.out.println("Arrray after insertion is:  ");
        for (int num : arr) {
            System.out.print(num+ " ");
            
        }
    }
}

//Linear Search: Find the target element is there or not
public class Main {

    public static void main(String[] args) {
        int[] arr={10,20,30,87,76};

        int target=67;

        boolean found=false;
        for (int i = 0; i < arr.length; i++) {
            if (arr[i]==target) {
                System.out.println("Target found at index "+ i);
                found=true;
                break;
                
            }
            
        }
        if (!found) {
            System.out.println("Target: "+ target+" Not found");
            
        }
    }
}

//pseudo Dynamic Array

public class Main {

    public static void main(String[] args) {
        int[] numbers = new int[3];
        numbers[0] = 10;
        numbers[1] = 20;
        numbers[2] = 30;
        System.out.println(numbers.length);
        for (int i = 0; i < numbers.length; i++) {
            System.out.print(numbers[i] + " ");

        }

        int[] newNumbers = new int[numbers.length * 2];
        for (int i = 0; i < numbers.length; i++) {
            newNumbers[i] = numbers[i];

        }
        newNumbers[3] = 40;
        newNumbers[4] = 50;
        newNumbers[5] = 60;

        numbers = newNumbers;
        System.out.println();

        for (int i = 0; i < numbers.length; i++) {
            System.out.print(numbers[i] + " ");

        }
        System.out.println();
        System.out.println(numbers.length);

    }
}

//TC -> O(n)

public class Main {

    public static void main(String[] args) {
        int[] arr = {5, 6, 8, 10, 12, 13, 52, 1};
        int n = arr.length;
        int max = arr[0];
        int min = arr[0];
        for (int num : arr) {
            if(num > max) {
                max = num;
            }
            if(num < min) {
                min = num;
            }
        }
        System.out.println("The maximum number is: " + max);
        System.out.println("The minimum number is: " + min);
    }
}

import java.util.Arrays;

public class Main {

    public static void main(String[] args) {
        int [] arr = {4, 5, 6, 203, 483, 55, 34};
        for (int i = 0,  j=arr.length-1; i < j; i++, j--) {
            int temp = arr[i];
            arr[i] = arr[j];
            arr[j] = temp;
        }
        System.out.print("The reserved array is: " + Arrays.toString(arr));
        
        
    }
}

public class Main {

    public static void main(String[] args) {
        int [] arr = {4, 5, 6, 203, 483, 55, 34};
        int i = 0,  j=arr.length-1;
        while(i < j) {
            int temp = arr[i];
            arr[i] = arr[j];
            arr[j] = temp;

            i++;
            j--;
        }
        
        System.out.print("The reserved array is: " + Arrays.toString(arr));
        
        
    }
}

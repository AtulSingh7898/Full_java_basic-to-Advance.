public class Main {

    public static int[] createArray(int size) {
        int[] arr = new int[size];
        for (int i = 0; i < arr.length; i++) {
            arr[i] = i+1;
        }
        return arr;
    }
    public static void main(String[] args) {
        int [] newArray = createArray(5);

        for (int num : newArray) {
            System.out.print(num + " ");
        }
    }
}

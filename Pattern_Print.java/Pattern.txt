    static void Printcolom(int col, int max, int row, int maxcol){
        if(col > max){
            return;

        if(col == 1 || col == max){
            System.out.print("* ");
        }else{
            System.out.print("  ");
        }
        
        Printcolom(col+1, max, row, col);
    }
    static void printSpace(int col, int max){
        if(col > max){
            return;
        }
        System.out.print(" ");
        printSpace(col+1, max);
    }

        if( row > 2*maxrow){
            return;

        printSpace(1, space);
        Printcolom(1, colom,row, maxrow);
        System.out.println();
        Printrows(row+1, maxrow);
        
    }
   
   
    public static void main(String[] args) {

    static void forStar(int min, int max, int promin, int promax){
        if(min > max) return;
        if(min == max || min == 1|| promin == 1 || 2*promax == promin || min == max/2){
            System.out.print("* ");
        }else{
            System.out.print("  ");
        }
        forStar(min+1, max, promin, promax);
    }

    static void pattern(int min, int max){
        if(min > 2*max) return;
        int space = (min > max) ? 2*max - min: min-1+1;
        int star = (min > max) ? min - max+1 : max - min+1;
        forSpace(1, space, min, max);
        forStar(1, star, min, max);
        System.out.println();
        pattern(min+1, max);
        
    }
    static void forSpace(int min, int max, int promin, int promax){
        if(min > max) return;
        if(promin == promax){
            System.out.print("* ");
        }else
        System.out.print(" ");
        forSpace(min+1, max, promin, promax);
    }

    public static void main(String[] args) {
        int num = 5;
        // pattern(1, num);

        for(int i = 1; i <= 2*num; i++){
            int space = (i > num) ? 2*num-i : i-1+1;
            int star = (i > num) ? i - num+1: num-i+1;
            for(int j = 1; j<= space; j++){
               if(i == num){
                    System.out.print("* ");
                }else{
                    System.out.print(" ");
                }
                
            }
            for(int j = 1; j <= star; j++){
            if(i == 1 || i == 2*num || j == 1 || j == star){
                    System.out.print("* ");
                }else{
                    System.out.print("  ");
                }
           }
           System.out.println();
        }
        
        
        
    }

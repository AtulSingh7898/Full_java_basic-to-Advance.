    public static void trianglePattern(int n){
        for(int i = 1; i <= n; i++){
            int count = 0;
            for(int j = 1; j <= i; j++){

                if((j+i)%2==0){
                    System.out.print("1 ");
                }else{
                    System.out.print("0 ");
                }
            }
            System.out.println();
        }
    }

        trianglePattern(5);
        
    }
